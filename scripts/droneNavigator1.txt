
class PIDController {
    constructor(kp = 1, ki = 0, kd = 0, minOutput = null, maxOutput = null, integralL = null, integralH = null) {
        this.kp = kp;
        this.ki = ki;
        this.kd = kd;
        this.minOutput = minOutput;
        this.maxOutput = maxOutput;
        this.integralLow = integralL || minOutput
        this.integralHigh = integralH || maxOutput
        
        this.integral = 0;
        this.lastError = 0;
    }

    calculate(error, currentKi = this.ki) {
        // Anti-Windup Triggered from the outside loop conditional
        if (currentKi === 0) {
            this.integral = 0;
        }

        const p = error * this.kp;
        const i = this.integral * currentKi;
        const d = (error - this.lastError) * this.kd;

        this.integral += error;
        this.lastError = error;

        let output = p + i + d;
        const originalOutput = output
        if (this.maxOutput !== null) output = Math.min(this.maxOutput, output);
        if (this.minOutput !== null) output = Math.max(this.minOutput, output);
        if (output != originalOutput || error < this.integralLow || error > this.integralHigh) {
            this.integral = 0
        }

        return output;
    }

    reset() {
        this.integral = 0;
        this.lastError = 0;
    }
}



// Instantiate the independent loops with your working limits and parameters
const altController   = new PIDController(0.04,    0.0004, 0.6, -0.1, 0.4, -5, 5);
const pitchController = new PIDController(0.001,    0.00001, 0.02, -0.1, 0.1);
const rollController  = new PIDController(0.001,    0.00001, 0.02, -0.1, 0.1);
const yawController = new PIDController(0.004,    0.001, 0.04, -0.1, 0.1);
const sideslipController = new PIDController(100,    0, 10, -50, 50);
const distanceController = new PIDController(0.6,    0, 2, -50, 50);

let waypoints = [
    {x: 200, y: 80, z: 200},
    {x: -200, y: 80, z: 200},
    {x: -200, y: 80, z: -200},
    {x: 200, y: 80, z: -200},
]
let waypointIndex = 0

startLoop(async () => {
    // Live Data
    const pos = info.position
    const alt = pos.y;
    const currentPitch = -info.rotation.x; 
    const currentYaw = info.rotation.y
    const currentRoll = info.rotation.z;
    const waypoint = waypoints[waypointIndex]

    

    // API Control: Take-off phase vs. Cruise phase
    if (alt < 10) {
        const t = 0.2
        setAllThrottle(t,t,t,t)
        
    } else {

        let pitch = 0
        let roll = 0
        let yaw = 0
        let throttleBase = 0
        let targetPitch = 0
        let targetRoll = 0
        let targetYaw = 0
        const targetDisError = Math.sqrt(Math.pow(waypoint.z-pos.z,2)+Math.pow(waypoint.x-pos.x,2))

        const targetAltitude = waypoint.y
        
        let dir = Math.atan2(waypoint.z - pos.z, -(waypoint.x - pos.x)) * 180 / Math.PI
        dir = dir < -90 ? 270 + dir : dir-90
        targetYaw = dir

        let yawError = targetYaw - currentYaw
        if (yawError < -180) yawError += 360
        if (yawError > 180) yawError -= 360

        const altError = targetAltitude - alt; 
        let climb = altController.calculate(altError);
        throttleBase = climb

        const sideslipError = 0 - info.air_speed.x
        targetRoll = -sideslipController.calculate(sideslipError)

        targetPitch = -distanceController.calculate(targetDisError)
        

        const pitchError = targetPitch - currentPitch;
        pitch = -pitchController.calculate(pitchError);


        const rollError = targetRoll - currentRoll;
        roll = -rollController.calculate(rollError);


        yaw = yawController.calculate(yawError);
        
        const t0 = Math.min(Math.max(-1, throttleBase - pitch - roll - yaw), 1);
        const t1 = Math.min(Math.max(-1, throttleBase - pitch + roll + yaw), 1);
        const t2 = Math.min(Math.max(-1, throttleBase + pitch - roll + yaw), 1);
        const t3 = Math.min(Math.max(-1, throttleBase + pitch + roll - yaw), 1);
        setAllThrottle(t0,t1,t2,t3)


        if (targetDisError < 10) {
            log("reached target")
            waypointIndex = (waypointIndex + 1) % waypoints.length
        }


    }

}, 50);


function setAllThrottle(t0, t1, t2, t3) {
    setThrottle(0,t0)
    setThrottle(1,t1)
    setThrottle(2,t2)
    setThrottle(3,t3)
}
