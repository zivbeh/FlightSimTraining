
class PIDController {
    constructor(kp = 1, ki = 0, kd = 0, minOutput = null, maxOutput = null) {
        this.kp = kp;
        this.ki = ki;
        this.kd = kd;
        this.minOutput = minOutput;
        this.maxOutput = maxOutput;
        
        this.integral = 0;
        this.lastError = 0;
    }

    calculate(error, currentKi = this.ki) {
        // Anti-Windup Triggered from the outside loop conditional
        if (currentKi === 0) {
            this.integral = 0;
        }

        const p = error * this.kp;
        const i = this.integral * currentKi;
        const d = (error - this.lastError) * this.kd;

        this.integral += error;
        this.lastError = error;

        let output = p + i + d;
        const originalOutput = output
        if (this.maxOutput !== null) output = Math.min(this.maxOutput, output);
        if (this.minOutput !== null) output = Math.max(this.minOutput, output);
        if (output != originalOutput) {
            this.integral = 0
        }

        return output;
    }

    reset() {
        this.integral = 0;
        this.lastError = 0;
    }
}

// Instantiate the independent loops with your working limits and parameters
const altController   = new PIDController(2,    0.05, 5, -15, 30);
const pitchController = new PIDController(1,    0.02, 2,   -3,  20);
const rollController  = new PIDController(1.2,  0.01, 3,   -15, 15);
const directionController = new PIDController(2.4,    0.1, 3, -60, 60);

let waypoints = [
    {x: 200, y: 80, z: 200},
    {x: -200, y: 80, z: 200},
    {x: -200, y: 80, z: -200},
    {x: 200, y: 80, z: -200},
]
let waypointIndex = 0


startLoop(async () => {
    // Live Data
    const pos = info.position
    const alt = pos.y;
    const currentPitch = -info.rotation.x; 
    const currentYaw = info.rotation.y
    const currentRoll = info.rotation.z;
    const waypoint = waypoints[waypointIndex]

    // API Control: Take-off phase vs. Cruise phase
    if (alt < 10) {
        setThrottle(1);
        setElevators(-10); 
        setRoll(0);       
        
        altController.reset();
        pitchController.reset();
        rollController.reset();
        
    } else {
        setThrottle(0.5);

        const targetDisError = Math.sqrt(Math.pow(waypoint.z-pos.z,2)+Math.pow(waypoint.x-pos.x,2))

        const targetAltitude = waypoint.y
        
        let targetRoll = 0
        let dir = Math.atan2(waypoint.z - pos.z, -(waypoint.x - pos.x)) * 180 / Math.PI
        dir = dir < -90 ? 270 + dir : dir-90

        let yawError = dir - currentYaw
        if (yawError < -180) yawError += 360
        if (yawError > 180) yawError -= 360
        targetRoll = directionController.calculate(-yawError)
        if (targetDisError < 30) targetRoll *= targetDisError/30
        if (targetDisError < 150 && Math.abs(targetRoll) > 20) targetRoll *= -0.5

        // --- 1. OUTER LOOP: ALTITUDE PID ---
        const altError = targetAltitude - alt; 
        let targetPitch = altController.calculate(altError);

        // --- 2. INNER LOOP: PITCH / ELEVATOR PID ---
        const pitchError = targetPitch - currentPitch;
        let elev = -pitchController.calculate(pitchError);
        elev += -Math.abs(currentRoll)/8
        elev = Math.min(elev, 20)
        setElevators(elev);

        // --- 3. ROLL / AILERON PID ---
        const rollError = targetRoll - currentRoll;
        let rollOutput = -rollController.calculate(rollError);
        setRoll(rollOutput);
        

        if (targetDisError < 10) {
            log("reached target")
            waypointIndex = (waypointIndex + 1) % waypoints.length
        }


    }

}, 50);



function setRoll(angle) {
    setAileronRight(angle);
    setAileronLeft(-angle);
}

function setElevators(angle) {
    setElevatorLeft(angle);
    setElevatorRight(angle);
}
